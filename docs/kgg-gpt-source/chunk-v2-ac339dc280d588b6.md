<script id="kgg-v096-cockpit-tablet-thirds-input">
(function(){
  "use strict";
  const PATCH_ID="kgg-v096-cockpit-tablet-thirds-input";
  const TABLET_QUERY="(min-width:760px)";
  var api=null,root=null,board=null,installed=false,boardObserver=null,bodyObserver=null,appStyleSnapshot=null;

  function tablet(){return !!(window.matchMedia&&window.matchMedia(TABLET_QUERY).matches);}
  function stateCount(){
    if(!api||typeof api.getState!=="function")return 0;
    try{
      var state=api.getState()||{};
      if(Number.isFinite(Number(state.slotCount)))return Math.max(0,Math.min(3,Number(state.slotCount)));
      return Array.isArray(state.slots)?state.slots.filter(Boolean).length:0;
    }catch(error){return 0;}
  }
  function syncAppPane(active,count){
    if(active&&document.body.classList.contains("kggTcSeamDragging"))return;
    var app=document.querySelector(".app");if(!app)return;
    var names=["position","left","right","width","max-width","min-width","box-sizing","height","min-height","max-height","margin","padding","overflow","overflow-x","transform","z-index"];
    if(active&&count>0&&count<3){
      if(!appStyleSnapshot)appStyleSnapshot=names.map(function(name){return {name:name,value:app.style.getPropertyValue(name),priority:app.style.getPropertyPriority(name)};});
      var viewportWidth=document.documentElement.clientWidth||window.innerWidth||0;
      app.style.setProperty("position","fixed","important");app.style.setProperty("left","0","important");
      app.style.setProperty("right",(viewportWidth*(count===1?1/3:2/3))+"px","important");
      app.style.setProperty("width","auto","important");app.style.setProperty("max-width","none","important");
      app.style.setProperty("min-width","0","important");app.style.setProperty("box-sizing","border-box","important");
      app.style.setProperty("height","100dvh","important");app.style.setProperty("min-height","0","important");app.style.setProperty("max-height","100dvh","important");
      app.style.setProperty("margin","0","important");app.style.setProperty("padding","10px","important");app.style.setProperty("overflow","auto","important");
      app.style.setProperty("overflow-x","hidden","important");app.style.setProperty("transform","none","important");app.style.setProperty("z-index","2147480999","important");
    }else if(appStyleSnapshot){
      appStyleSnapshot.forEach(function(item){if(item.value)app.style.setProperty(item.name,item.value,item.priority);else app.style.removeProperty(item.name);});appStyleSnapshot=null;
    }
  }
  function syncLayout(){
    if(!document.body)return;
    var count=stateCount(),active=tablet()&&document.body.classList.contains("kggTherapyCockpitOpen")&&count>0;
    document.body.classList.toggle("kggTcTabletSplit",active);
    document.body.classList.toggle("kggTcSlots1",active&&count===1);
    document.body.classList.toggle("kggTcSlots2",active&&count===2);
    document.body.classList.toggle("kggTcSlots3",active&&count===3);
    if(active)document.body.dataset.kggTcSlotCount=String(count);else delete document.body.dataset.kggTcSlotCount;
    syncAppPane(active,count);
    if(!tablet())undecorateRows();
  }
  function rowContext(row){
    try{
      var card=row.closest(".kgg-tc-card"),exercise=row.closest(".kgg-tc-exercise");
      if(!card||!exercise||!api||typeof api.getSlot!=="function")return null;
      var slotIndex=Number(card.getAttribute("data-tc-card")),slot=api.getSlot(slotIndex);
      var exerciseIndex=Array.prototype.indexOf.call(card.querySelectorAll(".kgg-tc-exercise"),exercise);
      var setLabel=row.querySelector(".kgg-tc-set-no"),match=String(setLabel&&setLabel.textContent||"").match(/\d+/);
      var setIndex=match?Number(match[0])-1:-1,ex=slot&&slot.exercises&&slot.exercises[exerciseIndex];
      return ex&&setIndex>=0?{ex:ex,setIndex:setIndex}:null;
    }catch(error){return null;}
  }
  function valueNode(className,value){var node=document.createElement("span");node.className=className;node.textContent=String(value==null||value===""?"—":value);return node;}
  function decorateRow(row){
    var context=rowContext(row);if(!context)return;
    var ex=context.ex,index=context.setIndex,previous=ex.previous&&ex.previous[index]||["",""];
    var loadUnit=String(ex.loadUnit||ex.weightUnit||"kg"),metricUnit=String(ex.metricUnit||ex.unit||"Wdh");
    var previousNode=row.querySelector(".kgg-tc-prev");
    if(previousNode&&!previousNode.classList.contains("kgg-v096-pair")){
      previousNode.classList.add("kgg-v096-pair");
      previousNode.replaceChildren(valueNode("kgg-tc-prev-value",previous[0]),valueNode("kgg-tc-prev-unit",loadUnit),valueNode("kgg-tc-prev-value",previous[1]),valueNode("kgg-tc-prev-unit",metricUnit));
      previousNode.setAttribute("aria-label","Vorwert: "+(previous[0]||"kein Wert")+" "+loadUnit+", "+(previous[1]||"kein Wert")+" "+metricUnit);
    }
    row.querySelectorAll(".kgg-tc-field").forEach(function(button){
      var unit=button.getAttribute("data-tc-field")==="load"?loadUnit:metricUnit;
      if(button.dataset.kggV096Unit===unit&&button.querySelector(".kgg-tc-field-unit"))return;
      var value=String(button.textContent||"").trim(),label="Heute "+(value||"kein Wert")+" "+unit;
      button.replaceChildren(valueNode("kgg-tc-field-value",value),valueNode("kgg-tc-field-unit",unit));
      button.dataset.kggV096Unit=unit;
      button.setAttribute("aria-label",label);
    });
  }
  function decorateRows(){
    if(!tablet()||!document.body.classList.contains("kggTcTabletSplit")||!board)return;
    board.querySelectorAll(".kgg-tc-set-row").forEach(decorateRow);
  }
  function undecorateRows(){
    if(!board)return;
    board.querySelectorAll(".kgg-tc-set-row").forEach(function(row){
      var context=rowContext(row);if(!context)return;
      var ex=context.ex,index=context.setIndex,previous=ex.previous&&ex.previous[index]||["",""];
      var prev=row.querySelector(".kgg-tc-prev");
      if(prev&&prev.classList.contains("kgg-v096-pair")){
        prev.classList.remove("kgg-v096-pair");
        prev.textContent=(previous[0]||"–")+" "+(ex.loadUnit||"")+" / "+(previous[1]||"–")+" "+(ex.metricUnit||"");
        prev.setAttribute("aria-label","Vorwert");
      }
      row.querySelectorAll(".kgg-tc-field").forEach(function(button){
        var field=button.getAttribute("data-tc-field"),value=ex.today&&ex.today[index]&&(field==="load"?ex.today[index][0]:ex.today[index][1]);
        if(button.dataset.kggV096Unit){button.textContent=String(value||"");delete button.dataset.kggV096Unit;}
      });
    });
  }
  function install(){
    if(installed)return;
    api=window.KGGTherapyCockpit;root=document.getElementById("kggTherapyCockpitRoot");board=document.getElementById("kggTherapyCockpitBoard");
    if(!api||!root||!board||!document.body){window.setTimeout(install,50);return;}
    installed=true;syncLayout();decorateRows();
    boardObserver=new MutationObserver(function(){syncLayout();decorateRows();});
    boardObserver.observe(board,{childList:true,subtree:true});
    bodyObserver=new MutationObserver(syncLayout);
    bodyObserver.observe(document.body,{attributes:true,attributeFilter:["class"]});
    window.addEventListener("resize",function(){syncLayout();decorateRows();},{passive:true});
    window.addEventListener("orientationchange",function(){window.setTimeout(function(){syncLayout();decorateRows();},180);},{passive:true});
    window.KGG_PATCHES=window.KGG_PATCHES||{};
    window.KGG_PATCHES[PATCH_ID]={installed:true,version:1,dependsOn:"kgg-v095-cockpit-responsive-entry",contract:"tablet slot widths are one third of viewport per patient; set number, previous pair and both today values stay on one row with visible units; keypad spans below the active set"};
  }
  if(document.readyState==="loading")document.addEventListener("DOMContentLoaded",install,{once:true});else install();
