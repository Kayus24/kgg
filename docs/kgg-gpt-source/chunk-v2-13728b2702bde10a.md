<!-- Kleine Plan-Karte zum Hinzufügen einer weiteren Übung -->
<script id="kgg-v091-plan-add-exercise-card">
(function(){
  "use strict";
  var PATCH_ID="kgg-v091-plan-add-exercise-card";
  var CARD_ID="kggPlanAddExerciseCard";
  var installed=false;

  function focusExerciseComposer(){
    var bankToggle=document.getElementById("bankToggle");
    if(bankToggle&&bankToggle.getAttribute("aria-label")==="Übungsdatenbank öffnen")bankToggle.click();
    var input=document.getElementById("exerciseInput");
    if(!input)return;
    try{input.focus({preventScroll:true});}catch(error){input.focus();}
    try{input.scrollIntoView({behavior:"smooth",block:"center",inline:"nearest"});}catch(error){input.scrollIntoView(false);}
  }

  function removeCard(list){
    var old=document.getElementById(CARD_ID);
    if(old&&old.parentNode===list)old.parentNode.removeChild(old);
  }

  function decoratePlanList(){
    var list=document.getElementById("planList");
    if(!list)return;
    var hasExercise=!!list.querySelector(".planCard[data-plan-id]");
    if(!hasExercise){removeCard(list);return;}
    var card=document.getElementById(CARD_ID);
    if(!card){
      card=document.createElement("button");
      card.id=CARD_ID;
      card.type="button";
      card.className="planAddCard";
      card.setAttribute("aria-label","Neue Übung hinzufügen");
      card.innerHTML='<span class="planAddCardPlus" aria-hidden="true">+</span><span class="planAddCardText">Übung hinzufügen</span>';
      card.addEventListener("click",focusExerciseComposer);
    }
    if(card.parentNode!==list)list.appendChild(card);
  }

  function install(){
    if(installed)return;
    var list=document.getElementById("planList");
    if(!list){
      setTimeout(install,50);
      return;
    }
    installed=true;
    decoratePlanList();
    var observer=new MutationObserver(decoratePlanList);
    observer.observe(list,{childList:true});
    window.KGG_PATCHES=window.KGG_PATCHES||{};
    window.KGG_PATCHES[PATCH_ID]={installed:true,version:1,dependsOn:"kgg-v090-cockpit-exercises-only"};
  }

  if(document.readyState==="loading")document.addEventListener("DOMContentLoaded",install,{once:true});else install();
})();
</script>
