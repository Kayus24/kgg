<script id="kgg-v112-cockpit-main-pane-phone-layout">
(function(){
  "use strict";
  var active=false,lastKey="",style=null,scheduled=false;
  function splitSelectors(value){
    var list=[],start=0,depth=0;
    for(var i=0;i<value.length;i++){var c=value[i];if(c==="("||c==="[")depth++;else if(c===")"||c==="]")depth--;else if(c===","&&depth===0){list.push(value.slice(start,i).trim());start=i+1;}}
    list.push(value.slice(start).trim());return list;
  }
  function scoped(selector){
    if(/^(html|:root)\b/.test(selector)||/^body$/.test(selector))return null;
    if(/^body\b/.test(selector)){
      var depth=0,pos=-1;for(var i=0;i<selector.length;i++){var c=selector[i];if(c==="("||c==="[")depth++;else if(c===")"||c==="]")depth--;else if(/\s/.test(c)&&depth===0){pos=i;break;}}
      if(pos<0)return null;
      var head=selector.slice(0,pos).replace(/^body/,'body.kggTcMainPanePhone'),tail=selector.slice(pos).trim();
      return head+" "+(/^\.app\b/.test(tail)?tail.replace(/^\.app\b/,".app.app.app"):".app.app.app "+tail);
    }
    return "body.kggTcMainPanePhone "+(/^\.app\b/.test(selector)?selector.replace(/^\.app\b/,".app.app.app"):".app.app.app "+selector);
  }
  function phoneMedia(condition,width,height){
    var query=condition.replace(/\((min|max)-width\s*:\s*([\d.]+)px\)/gi,function(_,bound,size){return (bound==="max"?width<=Number(size):width>=Number(size))?"(min-width:0px)":"(min-width:999999px)";});
    query=query.replace(/\(orientation\s*:\s*(portrait|landscape)\)/gi,function(_,direction){return ((height>=width)===(direction==="portrait"))?"(min-width:0px)":"(min-width:999999px)";});
    return matchMedia(query).matches;
  }
  function mirrorPhoneRules(width,height){
    var css=[];
    function walk(rules,inPhone){for(var rule of Array.from(rules||[])){
      if(rule.type===4){var matches=Array.from(rule.conditionText.matchAll(/max-width\s*:\s*([\d.]+)px/gi));var phone=inPhone||matches.some(function(match){return Number(match[1])<=759;});if(phoneMedia(rule.conditionText,width,height))walk(rule.cssRules,phone);}
      else if(rule.type===1&&inPhone){var selectors=splitSelectors(rule.selectorText).map(scoped).filter(Boolean);if(!selectors.length)continue;var declarations=Array.from(rule.style).map(function(name){var value=rule.style.getPropertyValue(name).replace(/(-?[\d.]+)vw\b/g,function(_,number){return Number(number)*width/100+"px";});return name+":"+value+"!important";}).join(";");css.push(selectors.join(",")+"{"+declarations+"}");}
      else if(rule.type===12&&CSS.supports(rule.conditionText))walk(rule.cssRules,inPhone);
    }}
    for(var sheet of Array.from(document.styleSheets)){if(sheet.ownerNode===style)continue;try{walk(sheet.cssRules,false);}catch(ignore){}}
    if(!style){style=document.createElement("style");style.id="kgg-v112-cockpit-main-pane-phone-layout-phone-rules";document.head.appendChild(style);}
    style.textContent=css.join("\n");
  }
  function sync(){
    scheduled=false;if(document.body.classList.contains("kggTcSeamDragging"))return;var app=document.querySelector(".app");if(!app)return;var split=matchMedia("(min-width:760px)").matches&&document.body.classList.contains("kggTherapyCockpitOpen")&&document.body.classList.contains("kggTcTabletSplit");document.body.classList.toggle("kggTcMainPaneCompact",split&&app.clientWidth<=580);document.body.classList.toggle("kggTcMainPaneVeryCompact",split&&app.clientWidth<=360);
    var next=matchMedia("(min-width:760px)").matches&&document.body.classList.contains("kggTherapyCockpitOpen")&&document.body.classList.contains("kggTcSlots2");
    var changed=next!==active;active=next;document.body.classList.toggle("kggTcMainPanePhone",active);
    if(!active){lastKey="";if(style)style.textContent="";}
    else{var key=Math.round(app.clientWidth)+":"+app.clientHeight;if(key!==lastKey||changed){lastKey=key;mirrorPhoneRules(app.clientWidth,app.clientHeight);}}
    if(changed){
      if(typeof window.KGGMainPaneLayoutRefresh==="function")window.KGGMainPaneLayoutRefresh();
      for(var key of ["KGG_UI_MINI_SERIES","KGG_UI_MINI_SERIES_V042","KGG_UI_PHONE_DRAWER_BANK_ALIGN_V045","KGG_UI_PHONE_MINI_FIX_V050"]){var helper=window[key];if(helper&&typeof helper.install==="function")helper.install();}
    }
  }
  function schedule(){if(scheduled)return;scheduled=true;requestAnimationFrame(sync);}
  new MutationObserver(schedule).observe(document.body,{attributes:true,attributeFilter:["class"]});
  var app=document.querySelector(".app");if(app&&window.ResizeObserver)new ResizeObserver(schedule).observe(app);
  addEventListener("resize",schedule,{passive:true});sync();
  window.KGG_PATCHES=window.KGG_PATCHES||{};
  window.KGG_PATCHES["kgg-v112-cockpit-main-pane-phone-layout"]={installed:true,version:1,contract:"two Cockpit slots select existing main-app phone presentation and handlers; phone CSS is scoped to the main pane, while Cockpit keeps its tablet layout; narrow cards preserve separate 44px actions"};
})();
