<script id="kgg-v120-cockpit-keypad-visibility">
(function(){
  "use strict";
  const PATCH_ID="kgg-v120-cockpit-keypad-visibility";
  function sync(){
    const pad=document.querySelector('.kgg-tc-pad');if(!pad)return;
    const card=pad.closest('.kgg-tc-card'),scroll=card&&card.querySelector('.kgg-tc-scroll'),active=card&&card.querySelector('.kgg-tc-field.active');if(!scroll||!active)return;
    const r=card.getBoundingClientRect(),vv=window.visualViewport;
    const floating=pad.offsetHeight+active.offsetHeight+16>scroll.clientHeight;
    pad.classList.toggle('kgg-tc-pad-floating',floating);
    if(floating){pad.style.left=(r.left+8)+'px';pad.style.width=Math.max(160,r.width-16)+'px';const dock=document.getElementById('kggPhoneCockpitDock'),dockRect=dock&&dock.getBoundingClientRect(),dockTop=dockRect&&dock.getClientRects().length&&getComputedStyle(dock).visibility!=='hidden'&&dockRect.top<innerHeight?dockRect.top:innerHeight;pad.style.bottom=Math.max(8,innerHeight-((vv?vv.offsetTop+vv.height:innerHeight)-8),innerHeight-dockTop+8)+'px';let value=pad.querySelector('.kgg-tc-pad-current');if(!value){value=document.createElement('strong');value.className='kgg-tc-pad-current';pad.querySelector('.kgg-tc-pad-head>span').appendChild(value);}const next=active.textContent.trim();if(value.textContent!==next)value.textContent=next;}
    else{pad.style.removeProperty('left');pad.style.removeProperty('width');pad.style.removeProperty('bottom');}
    // Reveal the whole keypad rather than only the input above it.
    if(!floating)pad.scrollIntoView({block:'nearest',inline:'nearest'});
  }
  let scheduled=false;
  function schedule(){if(scheduled)return;scheduled=true;requestAnimationFrame(function(){scheduled=false;sync();});}
  function boot(){const board=document.getElementById('kggTherapyCockpitBoard');if(!board){setTimeout(boot,50);return;}new MutationObserver(schedule).observe(board,{childList:true,subtree:true});}
  window.KGGCockpitPadVisibility=sync;
  addEventListener('resize',schedule,{passive:true});if(window.visualViewport){visualViewport.addEventListener('resize',schedule,{passive:true});visualViewport.addEventListener('scroll',schedule,{passive:true});}
  if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',boot,{once:true});else boot();
  window.KGG_PATCHES=window.KGG_PATCHES||{};
  window.KGG_PATCHES[PATCH_ID]={installed:true};
})();
</script>
